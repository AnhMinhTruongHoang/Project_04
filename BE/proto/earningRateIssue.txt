7. KIỂM TRA ARTIST EARNING TRONG TIDB
============================================================

SELECT
    `id`,
    `listeningSessionId`,
    `listenerId`,
    `artistId`,
    `trackId`,
    `ratePerStream`,
    `amount`,
    `currency`,
    `status`,
    `earningDate`,
    `qualifiedAt`,
    `availableAt`,
    `createdAt`
FROM `artist_earnings`
ORDER BY `createdAt` DESC
LIMIT 10;
