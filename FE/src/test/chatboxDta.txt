const getUserName = (user?: any) => {
  const email = user?.email?.trim();
  const username = user?.username?.trim();
  const name = user?.name?.trim();
  const badNames = ["user", "social user", "unknown", "unknown user"];

  if (email) return email.split("@")[0];

export type ArtistLeaderboardItem = {
  _id?: string;
  id?: string;
  name: string;
  email?: string;
  username?: string;
  avatarUrl?: string;
  avatar?: string;
  role?: string;
  type?: string;
  followers?: number | null;
  following?: number | null;
  tracks?: number | null;
  trackCount?: number | null;
  totalTracks?: number | null;
};

export const getArtistLeaderboard = async (
  limit = 10
): Promise<ArtistLeaderboardItem[]> => {
  const res = await fetch(
    `${process.env.NEXT_PUBLIC_BACKEND_URL}/api/v1/users/leaderboard/artists?limit=${limit}`,
    { cache: "no-store" }
  );

  if (!res.ok) throw new Error("Failed to fetch artist leaderboard");

  const json = await res.json();
  return Array.isArray(json?.data) ? json.data : [];
};
