"use client";

import { useEffect, useRef } from "react";
import { useTrackContext } from "@/lib/track.wrapper";
import { useHasMounted } from "@/utils/customHook";

import AppBar from "@mui/material/AppBar";
import Box from "@mui/material/Box";
import IconButton from "@mui/material/IconButton";
import Typography from "@mui/material/Typography";

import FavoriteBorderRoundedIcon from "@mui/icons-material/FavoriteBorderRounded";
import PersonAddAltRoundedIcon from "@mui/icons-material/PersonAddAltRounded";
import QueueMusicRoundedIcon from "@mui/icons-material/QueueMusicRounded";
import MoreHorizRoundedIcon from "@mui/icons-material/MoreHorizRounded";

import AudioPlayer from "react-h5-audio-player";
import "react-h5-audio-player/lib/styles.css";

const BACKEND_URL = process.env.NEXT_PUBLIC_BACKEND_URL || "";

const getAudioSrc = (trackUrl?: string) => {
  if (!trackUrl) return "/audio/DemoS.mp3";
  if (trackUrl.startsWith("http")) return trackUrl;
  if (trackUrl.startsWith("/")) return trackUrl;
  return `${BACKEND_URL}/tracks/${trackUrl}`;
};

const getImageSrc = (imgUrl?: string) => {
  if (!imgUrl) return "/audio/SC.png";
  if (imgUrl.startsWith("http")) return imgUrl;
  if (imgUrl.startsWith("/")) return imgUrl;
  return `${BACKEND_URL}/images/${imgUrl}`;
};

const AppFooter = () => {
  const hasMounted = useHasMounted();
  const playerRef = useRef<any>(null);
  const lastTrackIdRef = useRef<string>("");

  const { currentTrack, setCurrentTrack } = useTrackContext() as ITrackContext;

  useEffect(() => {
    const audio = playerRef.current?.audio?.current;
    if (!audio) return;

    if (currentTrack?._id && lastTrackIdRef.current !== currentTrack._id) {
      audio.currentTime = 0;
      lastTrackIdRef.current = currentTrack._id;
    }

    if (currentTrack?.isPlaying) {
      audio.play().catch(() => {});
    } else {
      audio.pause();
    }
  }, [currentTrack?._id, currentTrack?.isPlaying]);

  if (!hasMounted) return null;
  if (!currentTrack?._id) return null;

  return (
    <AppBar
      position="fixed"
      elevation={0}
      sx={{
        top: "auto",
        bottom: 0,
        height: 54,
        backgroundColor: "#1b1d1e",
        borderTop: "1px solid rgba(255,255,255,0.08)",
        boxShadow: "0 -8px 28px rgba(0,0,0,0.35)",
        zIndex: 1300,

        ".rhap_container": {
          width: "100%",
          height: 54,
          p: 0,
          backgroundColor: "transparent",
          boxShadow: "none",
          color: "#ffffff",
        },

        ".rhap_main": {
          height: "100%",
          display: "flex",
          alignItems: "center",
          gap: "14px",
        },

        ".rhap_progress-section": {
          flex: "1 1 auto",
          minWidth: 0,
          display: "flex",
          alignItems: "center",
          gap: "10px",
        },

        ".rhap_controls-section": {
          flex: "0 0 230px",
          m: 0,
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          gap: "8px",
        },

        ".rhap_main-controls": {
          display: "flex",
          alignItems: "center",
          gap: "6px",
        },

        ".rhap_main-controls-button": {
          m: 0,
          color: "#f2f2f2",
          width: 32,
          height: 32,
          fontSize: 26,
          transition: "0.15s ease",
          "&:hover": {
            color: "#ffffff",
          },
        },

        ".rhap_play-pause-button": {
          width: 34,
          height: 34,
          borderRadius: "50%",
          backgroundColor: "#2b2d2f",
          color: "#ffffff",
          fontSize: 28,
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          "&:hover": {
            backgroundColor: "#3a3d40",
          },
        },

        ".rhap_skip-button": {
          color: "#f2f2f2",
          fontSize: 24,
          width: 28,
          height: 28,
        },

        ".rhap_progress-container": {
          m: 0,
          height: 16,
          display: "flex",
          alignItems: "center",
        },

        ".rhap_progress-bar": {
          height: 2,
          backgroundColor: "#3a3d40",
          borderRadius: 99,
        },

        ".rhap_download-progress": {
          height: 2,
          backgroundColor: "#4a4d50",
          borderRadius: 99,
        },

        ".rhap_progress-filled": {
          height: 2,
          backgroundColor: "#ff5500",
          borderRadius: 99,
        },

        ".rhap_progress-indicator": {
          width: 10,
          height: 10,
          top: -4,
          marginLeft: -5,
          backgroundColor: "#ff5500",
          boxShadow: "0 0 0 4px rgba(255,85,0,0.16)",
        },

        ".rhap_time": {
          color: "#c9c9c9",
          fontSize: 11,
          fontWeight: 800,
          minWidth: 36,
        },

        ".rhap_volume-container": {
          flex: "0 0 auto",
          marginRight: 10,
        },

        ".rhap_volume-button": {
          color: "#f2f2f2",
          fontSize: 20,
          width: 26,
          height: 26,
          m: 0,
        },

        ".rhap_volume-bar": {
          backgroundColor: "#3a3d40",
        },

        ".rhap_volume-indicator": {
          backgroundColor: "#ffffff",
        },
      }}
    >
      <Box
        sx={{
          width: "100%",
          maxWidth: 1240,
          mx: "auto",
          height: 54,
          px: { xs: 1, md: 2 },
          display: "flex",
          alignItems: "center",
          gap: 2,
          backgroundColor: "#1b1d1e",
        }}
      >
        <Box
          sx={{
            flex: 1,
            minWidth: 0,
          }}
        >
          <AudioPlayer
            ref={playerRef}
            src={getAudioSrc(currentTrack.trackUrl)}
            volume={0.5}
            layout="horizontal-reverse"
            showSkipControls
            showJumpControls={false}
            autoPlayAfterSrcChange={false}
            onPlay={() => {
              setCurrentTrack({ ...currentTrack, isPlaying: true });
            }}
            onPause={() => {
              setCurrentTrack({ ...currentTrack, isPlaying: false });
            }}
            onEnded={() => {
              setCurrentTrack({ ...currentTrack, isPlaying: false });
            }}
          />
        </Box>

        <Box
          sx={{
            width: { xs: 0, sm: 260 },
            display: { xs: "none", sm: "flex" },
            alignItems: "center",
            gap: 1.1,
            minWidth: 0,
            borderLeft: "1px solid rgba(255,255,255,0.08)",
            pl: 1.5,
          }}
        >
          <Box
            component="img"
            src={getImageSrc(currentTrack.imgUrl)}
            alt={currentTrack.title}
            onError={(e) => {
              e.currentTarget.src = "/audio/SC.png";
            }}
            sx={{
              width: 34,
              height: 34,
              objectFit: "cover",
              borderRadius: "2px",
              backgroundColor: "#111",
              flexShrink: 0,
            }}
          />

          <Box sx={{ minWidth: 0, flex: 1 }}>
            <Typography
              noWrap
              title={currentTrack.description}
              sx={{
                color: "#a7a7a7",
                fontSize: 11,
                lineHeight: 1.2,
                fontWeight: 700,
              }}
            >
              {currentTrack.description ||
                currentTrack.uploader?.name ||
                "Unknown artist"}
            </Typography>

            <Typography
              noWrap
              title={currentTrack.title}
              sx={{
                color: "#ffffff",
                fontSize: 12,
                lineHeight: 1.3,
                fontWeight: 900,
                mt: 0.2,
              }}
            >
              {currentTrack.title}
            </Typography>
          </Box>
        </Box>

        <Box
          sx={{
            display: { xs: "none", md: "flex" },
            alignItems: "center",
            gap: 0.5,
          }}
        >
          {[
            FavoriteBorderRoundedIcon,
            PersonAddAltRoundedIcon,
            QueueMusicRoundedIcon,
            MoreHorizRoundedIcon,
          ].map((Icon, index) => (
            <IconButton
              key={index}
              size="small"
              sx={{
                width: 30,
                height: 30,
                color: "#d6d6d6",
                "&:hover": {
                  color: "#ffffff",
                  backgroundColor: "rgba(255,255,255,0.08)",
                },
              }}
            >
              <Icon sx={{ fontSize: 19 }} />
            </IconButton>
          ))}
        </Box>
      </Box>
    </AppBar>
  );
};

export default AppFooter;
